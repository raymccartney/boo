this.frameIndex = 0; } } document.addEventListener("DOMContentLoaded", () => { // Inject a hidden diagnostic visual logger to the UI const logDiv = document.createElement('div'); logDiv.style = "position:absolute; for (let i = event.resultIndex; i < event.results.length; ++i) { if (event.results[i].isFinal) { finalTranscript += event.results[i][0].transcript; } else { interimTranscript += event.results[i][0].transcript; } } const currentSpeech = (finalTranscript || interimTranscript).toLowerCase(); logDiv.innerText = "Heard: " + currentSpeech; // Trigger on fuzzy semantic fragments so background static noise won't break things if (currentSpeech.includes('garden') || currentSpeech.includes('gard')) { logDiv.innerText = "Executing: MOVE TO GARDEN"; }, shortestTravelTime); } }; recognition.onend = () => { recognition.start(); }; document.getElementById('tap-overlay').addEventListener('click', (e) => { recognition.start(); document.getElementById('tap-overlay').style.display = 'none';
